-- pagos del rango agrupados en la base (por la fecha del pago, en Bogotá)
SELECT estado, metodo, tipo_vehiculo, count(*) AS cantidad, sum(monto) AS total
      FROM pago WHERE (fecha AT TIME ZONE 'America/Bogota')::date BETWEEN $1::date AND $1::date GROUP BY estado, metodo, tipo_vehiculo ORDER BY estado, metodo, tipo_vehiculo;
-- parámetros: ["2026-11-20"]
