-- periodos de PA08
SELECT m.id, m.fecha_inicio::text AS inicio, m.fecha_fin::text AS fin, p.comprobante, p.estado, p.meses, p.monto, p.metodo
    FROM mensualidad m LEFT JOIN pago p ON p.id = m.pago_id WHERE m.vehiculo_id = $1 ORDER BY m.fecha_inicio;
