-- registro del cliente
SELECT c.id, c.tipo_documento, c.documento, c.nombre, c.telefono, c.correo, c.registrado_en, u.nombre_usuario AS registrado_por
      FROM cliente c LEFT JOIN cuenta_usuario u ON u.id = c.registrado_por_cuenta_id WHERE c.documento = $1;
